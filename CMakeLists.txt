cmake_minimum_required(VERSION 3.20)
project(vega CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -O3 -march=native)
find_package(Threads REQUIRED)
add_library(vega INTERFACE)
target_include_directories(vega INTERFACE include)
enable_testing()
add_executable(test_book tests/test_book.cpp)
target_link_libraries(test_book vega Threads::Threads)
add_test(NAME test_book COMMAND test_book)
add_executable(bench_book bench/bench_book.cpp)
target_link_libraries(bench_book vega)
